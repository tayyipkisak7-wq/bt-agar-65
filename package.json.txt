<script>
    let ws;

    function connectGame() {
        const nickname = document.getElementById('nick').value;
        const protocol = location.protocol === 'https:' ? 'wss://' : 'ws://';
        const wsUrl = protocol + location.host;

        ws = new WebSocket(wsUrl);

        ws.onopen = function() {
            document.getElementById('status').innerText = 'Sunucuya başarıyla bağlandı: ' + nickname;
            console.log('WebSocket bağlantısı kuruldu.');
            // Oyuncu giriş paketini sunucuya gönder
            ws.send(JSON.stringify({ type: 'join', name: nickname }));
        };

        ws.onmessage = function(event) {
            console.log('Sunucudan mesaj alındı:', event.data);
        };

        ws.onclose = function() {
            document.getElementById('status').innerText = 'Bağlantı kesildi.';
            document.getElementById('status').style.color = 'red';
        };
    }

    // Overlay kaldırma
    document.querySelectorAll("#gamemodes li").forEach(function(li){
        li.addEventListener("click", function(){
            document.getElementById("overlay").style.display = "none";
        });
    });
</script>
